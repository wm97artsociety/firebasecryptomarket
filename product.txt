    if (window.ethereum) {
      provider = new ethers.providers.Web3Provider(window.ethereum, "any");
      await provider.send("eth_requestAccounts", []);
      signer = provider.getSigner();

      updateWalletDisplay(account);
      localStorage.setItem('walletAddress', account); // Store connected wallet

    }

    }

      const productId = getProductIdFromUrl();

          if (!doc.exists) {
              productDetailDiv.innerHTML = '<p style="color: red;">Product not found.</p>';
              return;
          }

          const product = doc.data();
          const usdValue = ethToUsd > 0 ? (product.ethPrice * ethToUsd).toFixed(2) : 'N/A';

          productDetailDiv.innerHTML = `
              <img src="${product.image || 'placeholder-image.png'}" alt="${product.name}" />
              <h3>${product.name}</h3>
              <p>${product.description}</p>
              <p><strong>Keywords:</strong> ${product.keywords ? product.keywords.join(', ') : 'None'}</p>
              <p class="price">${product.ethPrice} ETH (~$${usdValue})</p>
              ${socialNftHtml}
          `;

      }
  }
