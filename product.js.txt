  async function fetchEthToUsd() {
    try {
      const res = await fetch('https://api.coingecko.com/api/v3/simple/price?ids=ethereum&vs_currencies=usd');
      const data = await res.json();
      ethToUsd = parseFloat(data.ethereum.usd);
       // Update the displayed price with USD equivalent after fetching rate
      if (currentProduct && ethToUsd) {
          const usdValue = (currentProduct.ethPrice * ethToUsd).toFixed(2);
          const priceElement = document.querySelector('#productDetail .price');
          if(priceElement) {
              priceElement.textContent = `${currentProduct.ethPrice} ETH (~$${usdValue})`;
          }
      }
    } catch (err) {
      console.error('Error fetching ETH price:', err);
      ethToUsd = 0; // Set to 0 or handle error appropriately
      const priceElement = document.querySelector('#productDetail .price');
      if(priceElement && currentProduct) {
          priceElement.textContent = `${currentProduct.ethPrice} ETH (USD rate unavailable)`;
      }
    }
  }

  async function connectWallet() {
    if (window.ethereum) {
      try {
        provider = new ethers.providers.Web3Provider(window.ethereum);
        await provider.send("eth_requestAccounts", []); // Request accounts
        signer = provider.getSigner();
        userAccount = await signer.getAddress();
        document.getElementById('walletDisplay').textContent = `Wallet: ${userAccount.substring(0, 6)}...${userAccount.substring(userAccount.length - 4)}`;
        localStorage.setItem('walletAddress', userAccount); // Store wallet address

         // Check if admin button should be shown after connecting
        if (userAccount.toLowerCase() === ADMIN_WALLET.toLowerCase()) {
             document.getElementById('adminBtn').style.display = 'block';
        } else {
             document.getElementById('adminBtn').style.display = 'none';
        }

      } catch (error) {
        console.error('Error connecting wallet:', error);
        alert('Failed to connect wallet.');
      }
    } else {
      alert('MetaMask or compatible wallet not detected.');
    }
  }

  function disconnectWallet() {
      userAccount = null;
      provider = null;
      signer = null;
      document.getElementById('walletDisplay').textContent = 'Wallet: —';
      localStorage.removeItem('walletAddress'); // Clear stored wallet address
      document.getElementById('adminBtn').style.display = 'none'; // Hide admin button on disconnect
  }


  // Function to get product ID from URL
  function getProductIdFromUrl() {
    const params = new URLSearchParams(window.location.search);
    return params.get('id');
  }

  // Function to display product details (FETCHING FROM FIRESTORE)
  async function loadProduct() { // Renamed from displayProductDetail for clarity with product.js
      const productId = getProductIdFromUrl();
      const productDetailDiv = document.getElementById('productDetail');
      productDetailDiv.innerHTML = '<p>Loading product details...</p>'; // Loading indicator

      if (!productId) {
          productDetailDiv.innerHTML = '<p style="color: red;">Product ID not found in URL.</p>';
          return;
      }

      try {
          // Fetch product from Firestore by ID
          const doc = await db.collection('products').doc(productId).get();

          if (!doc.exists) {
              productDetailDiv.innerHTML = '<p style="color: red;">Product not found.</p>';
              return;
          }

          const product = doc.data();
          currentProduct = product; // Store the fetched product
          const usdValue = ethToUsd > 0 ? (product.ethPrice * ethToUsd).toFixed(2) : 'N/A';

          // Display social media/NFT info if available
          let socialNftHtml = '';
          if (product.socialPostUrl) {
              socialNftHtml += `<p><strong>Social Media:</strong> <a href="${product.socialPostUrl}" target="_blank">${product.socialPostUrl}</a></p>`;
          }
          if (product.nftContractAddress && product.nftTokenId) {
               socialNftHtml += `<p><strong>NFT Details:</strong> Contract: ${product.nftContractAddress?.substring(0, 6)}..., Token ID: ${product.nftTokenId}</p>`; // Added optional chaining
               // You would typically add logic here to fetch more NFT details using an API
          }

          productDetailDiv.innerHTML = `
              <img src="${product.image || 'placeholder-image.png'}" alt="${product.name}" />
              <h3>${product.name}</h3>
              <p>${product.description}</p>
              <p><strong>Keywords:</strong> ${product.keywords && product.keywords.length > 0 ? product.keywords.join(', ') : 'None'}</p> <p class="price">${product.ethPrice} ETH${usdValue !== 'N/A' ? ' (~$' + usdValue + ')' : ''}</p> ${socialNftHtml}
          `;

      } catch (error) {
          console.error("Error fetching product details:", error);
          productDetailDiv.innerHTML = '<p style="color: red;">Error loading product details.</p>';
      }
  }
